cmake_minimum_required(VERSION 3.21)

set(CMAKE_ANDROID_API 18)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Android")
    message(FATAL_ERROR "Butterscotch supports iOS and Android only. Configure with -DCMAKE_SYSTEM_NAME=iOS or an Android NDK toolchain. System name: ${CMAKE_SYSTEM_NAME}")
endif()

set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum iOS version")
if(CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "12.0")
    message(FATAL_ERROR
        "Butterscotch requires iOS 12.0 or newer "
        "(requested: ${CMAKE_OSX_DEPLOYMENT_TARGET})."
    )
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(BUTTERSCOTCH_PLATFORM "ios" CACHE STRING "Butterscotch platform" FORCE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Android")
    set(BUTTERSCOTCH_PLATFORM "android" CACHE STRING "Butterscotch platform" FORCE)
endif()
set(BACKEND "sdl2" CACHE STRING "iOS backend")
set(AUDIO_BACKEND "miniaudio" CACHE STRING "iOS audio backend")

project(butterscotch LANGUAGES C CXX)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJC OBJCXX)
endif()

if(NOT BACKEND STREQUAL "sdl2")
    message(FATAL_ERROR "iOS/Android requires the SDL2 backend.")
endif()

if(NOT AUDIO_BACKEND STREQUAL "miniaudio")
    message(FATAL_ERROR "iOS/Android requires the miniaudio backend.")
endif()

set(ENABLE_LEGACY_GL OFF CACHE BOOL "Enable the legacy OpenGL renderer")
set(ENABLE_MODERN_GL ON CACHE BOOL "Enable the modern OpenGL renderer")
set(ENABLE_METAL ON CACHE BOOL "Enable the Metal renderer")

if(ENABLE_LEGACY_GL)
    message(FATAL_ERROR "The legacy OpenGL renderer is not supported on iOS/Android.")
endif()

if(NOT ENABLE_MODERN_GL AND NOT ENABLE_METAL)
    message(FATAL_ERROR "Enable at least one of ENABLE_MODERN_GL or ENABLE_METAL.")
endif()

# iOS frameworks
if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    if(ENABLE_MODERN_GL)
        find_library(IOS_OPENGLES_FRAMEWORK OpenGLES REQUIRED)
    endif()
    find_library(IOS_METAL_FRAMEWORK Metal REQUIRED)
    find_library(IOS_UIKIT_FRAMEWORK UIKit REQUIRED)
    find_library(IOS_FOUNDATION_FRAMEWORK Foundation REQUIRED)
    find_library(IOS_PHOTOS_FRAMEWORK Photos REQUIRED)
    find_library(IOS_GAMECONTROLLER_FRAMEWORK GameController)
    find_library(IOS_VIDEOTOOLBOX_FRAMEWORK VideoToolbox REQUIRED)
    find_library(IOS_Z_LIBRARY z REQUIRED)
    find_library(IOS_ICONV_LIBRARY iconv REQUIRED)
    find_library(IOS_ICUCORE_LIBRARY icucore REQUIRED)
endif()

# Compiler
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_compile_options(
    $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-Wall>
    $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-Wextra>
    $<$<COMPILE_LANGUAGE:C>:-Wvla>
)

if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(
        $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-ffast-math>
        $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-fno-finite-math-only>
    )

    include(CheckCCompilerFlag)
    check_c_compiler_flag("-fomit-frame-pointer" HAS_FOMIT_FRAME_POINTER)

    if(HAS_FOMIT_FRAME_POINTER)
        add_compile_options(
            $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-fomit-frame-pointer>
        )
    endif()
endif()

# Build metadata
find_package(Git QUIET)

set(BUTTERSCOTCH_COMMIT_HASH "unknown")
set(BUTTERSCOTCH_COMMIT_DATE "unknown")

if(GIT_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/.git")
    execute_process(
        COMMAND "${GIT_EXECUTABLE}" rev-parse --short HEAD
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        OUTPUT_VARIABLE BUTTERSCOTCH_COMMIT_HASH
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )

    execute_process(
        COMMAND "${GIT_EXECUTABLE}" log -1 --format=%cs
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        OUTPUT_VARIABLE BUTTERSCOTCH_COMMIT_DATE
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
endif()

message(STATUS "Git commit   : ${BUTTERSCOTCH_COMMIT_HASH}")
message(STATUS "Commit date  : ${BUTTERSCOTCH_COMMIT_DATE}")

# Make commit information available to C/C++
add_compile_definitions(
    BUTTERSCOTCH_COMMIT_HASH="${BUTTERSCOTCH_COMMIT_HASH}"
    BUTTERSCOTCH_COMMIT_DATE="${BUTTERSCOTCH_COMMIT_DATE}"
    $<$<BOOL:${ENABLE_MODERN_GL}>:ENABLE_GLES>
    $<$<BOOL:${ENABLE_MODERN_GL}>:ENABLE_MODERN_GL>
    $<$<BOOL:${ENABLE_METAL}>:ENABLE_METAL>
    GL_GLEXT_PROTOTYPES=1
    USE_MINIAUDIO
    USE_SDL2
)

if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_compile_definitions(
        BUTTERSCOTCH_PLATFORM_IOS
    )

    # Make commit information available to Swift
    set(BUILD_INFO_SWIFT "${CMAKE_CURRENT_BINARY_DIR}/BuildInfo.swift")

    file(WRITE "${BUILD_INFO_SWIFT}"
    "// Generated by CMake. Do not edit.\n"
    "enum BuildInfo {\n"
    "    static let commitHash = \"${BUTTERSCOTCH_COMMIT_HASH}\"\n"
    "    static let commitDate = \"${BUTTERSCOTCH_COMMIT_DATE}\"\n"
    "}\n"
    )
endif()

# WAD support
option(ENABLE_WAD14 "Enable support for WAD Version 14 and below" ON)
option(ENABLE_WAD16 "Enable support for WAD Version 15/16" ON)
option(ENABLE_WAD17 "Enable support for WAD Version 17" ON)
if(NOT ENABLE_WAD14 AND NOT ENABLE_WAD16 AND NOT ENABLE_WAD17)
    message(FATAL_ERROR "At least one WAD version must be enabled.")
endif()

foreach(WAD IN ITEMS 14 16 17)
    if(ENABLE_WAD${WAD})
        add_compile_definitions(ENABLE_WAD${WAD})
    endif()
endforeach()

# Sources
file(GLOB SOURCES
    src/*.c
    src/backends/sdl2.c
)

set(IOS_SOURCES src/ios/main.m src/ios/touch_overlay.m src/ios/zip_import.c)

file(GLOB ANDROID_SOURCES
    src/android/*.c
)

file(GLOB AUDIO_SOURCES
    src/audio/miniaudio/*.c
    src/audio/miniaudio/*.m
)

if(ENABLE_MODERN_GL)
    file(GLOB GL_SOURCES
        src/image/*.c
        src/gl_common/*.c
        src/gl/*.c
    )
    list(APPEND SOURCES ${GL_SOURCES})
endif()
if(ENABLE_METAL)
    if(NOT ENABLE_MODERN_GL)
        file(GLOB METAL_IMAGE_SOURCES src/image/*.c)
        list(APPEND SOURCES ${METAL_IMAGE_SOURCES})
    endif()
    list(APPEND SOURCES
        src/metal/metal_renderer.m
        src/metal/metal_draw.m
        src/metal/metal_surfaces.m
        src/metal/metal_state.m
        src/metal/metal_shaders.m
    )
endif()

# Dependencies
add_subdirectory(vendor/SDL2 ${CMAKE_BINARY_DIR}/SDL2)

if(NOT EXISTS "${CMAKE_SOURCE_DIR}/vendor/miniz/miniz.c")
    message(FATAL_ERROR "iOS/Android game import requires vendor/miniz.")
endif()

add_library(miniz STATIC
    vendor/miniz/miniz.c
    vendor/miniz/miniz_tdef.c
    vendor/miniz/miniz_tinfl.c
    vendor/miniz/miniz_zip.c
)

target_include_directories(miniz SYSTEM PUBLIC
    ${CMAKE_SOURCE_DIR}/vendor/miniz
)


add_library(stb_ds INTERFACE)
target_include_directories(stb_ds SYSTEM INTERFACE
    ${CMAKE_SOURCE_DIR}/vendor/stb/ds
)

add_library(stb_vorbis INTERFACE)
target_include_directories(stb_vorbis SYSTEM INTERFACE
    ${CMAKE_SOURCE_DIR}/vendor/stb/vorbis
)

add_library(sha1 STATIC vendor/sha1/sha1.c)
target_include_directories(sha1 SYSTEM PUBLIC
    ${CMAKE_SOURCE_DIR}/vendor/sha1
)
target_compile_options(sha1 PRIVATE -w)

add_library(bzip2 STATIC
    vendor/bzip2/bz_internal_error.c
    vendor/bzip2/bzlib.c
    vendor/bzip2/crctable.c
    vendor/bzip2/decompress.c
    vendor/bzip2/huffman.c
    vendor/bzip2/randtable.c
)

target_include_directories(bzip2 PUBLIC
    ${CMAKE_SOURCE_DIR}/vendor/bzip2
)
target_compile_options(bzip2 PRIVATE -w)

# Application
if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(butterscotch
        ${SOURCES}
        ${AUDIO_SOURCES}
        ${BUILD_INFO_SWIFT}
        vendor/md5/md5.c
        vendor/base64/base64.c
    )
else()
    add_library(butterscotch SHARED
        ${SOURCES}
        ${AUDIO_SOURCES}
        ${BUILD_INFO_SWIFT}
        vendor/md5/md5.c
        vendor/base64/base64.c
    )
    target_link_libraries(butterscotch PRIVATE
        android
        log
        GLESv2
        egl
    )
endif()

if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_sources(butterscotch PUBLIC ${IOS_SOURCES})
else()
    target_sources(butterscotch PUBLIC ${ANDROID_SOURCES})
endif()

target_include_directories(butterscotch PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/src/gl
    ${CMAKE_SOURCE_DIR}/src/gl_common
    ${CMAKE_SOURCE_DIR}/src/metal
    ${CMAKE_SOURCE_DIR}/src/image
    ${CMAKE_SOURCE_DIR}/src/audio/miniaudio
    ${SDL2_INCLUDE_DIRS}
)

if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_include_directories(butterscotch PRIVATE ${CMAKE_SOURCE_DIR}/src/ios)
endif()

if(ENABLE_MODERN_GL)
    target_include_directories(butterscotch PRIVATE
        ${CMAKE_OSX_SYSROOT}/System/Library/Frameworks/OpenGLES.framework/Headers
    )
endif()

target_include_directories(butterscotch SYSTEM PRIVATE
    ${CMAKE_SOURCE_DIR}/vendor/miniaudio
)

target_include_directories(butterscotch PUBLIC
    ${CMAKE_SOURCE_DIR}/vendor/md5
    ${CMAKE_SOURCE_DIR}/vendor/base64
    ${CMAKE_SOURCE_DIR}/vendor/stb/image
)

target_link_libraries(butterscotch PRIVATE
    SDL2-static
    miniz
    bzip2
    stb_ds
    stb_vorbis
    sha1
    $<$<BOOL:${ENABLE_MODERN_GL}>:${IOS_OPENGLES_FRAMEWORK}>
    $<$<BOOL:${ENABLE_METAL}>:${IOS_METAL_FRAMEWORK}>
)

if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(butterscotch PRIVATE
        ${IOS_UIKIT_FRAMEWORK}
        ${IOS_FOUNDATION_FRAMEWORK}
        ${IOS_PHOTOS_FRAMEWORK}
        ${IOS_VIDEOTOOLBOX_FRAMEWORK}
        ${IOS_Z_LIBRARY}
        ${IOS_ICONV_LIBRARY}
        ${IOS_ICUCORE_LIBRARY}
        pthread
    )

    if(IOS_GAMECONTROLLER_FRAMEWORK)
        target_link_libraries(butterscotch PRIVATE ${IOS_GAMECONTROLLER_FRAMEWORK})
    endif()
endif()

add_compile_options(
    $<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-w>
    $<$<COMPILE_LANGUAGE:Swift>:-suppress-warnings>
)


if(TARGET sdl_headers_copy)
    add_dependencies(butterscotch sdl_headers_copy)
endif()

# VM / diagnostics
option(ENABLE_VM_GML_PROFILER "Enable Butterscotch VM/interpreter profiler" OFF)
option(ENABLE_VM_TRACING "Enable Butterscotch VM/interpreter tracing checks" OFF)
option(ENABLE_VM_OPCODE_PROFILER "Enable Butterscotch VM opcode count profiler" OFF)
option(ENABLE_VM_STUB_LOGS "Enable VM unknown/stubbed function logs" OFF)
option(ENABLE_SPATIAL_GRID_LOGS "Enable Spatial Grid logs" OFF)
option(ENABLE_VM_EXCEPTIONS_LOGS "Enable VM Exceptions logs" OFF)

foreach(OPTION
    ENABLE_VM_GML_PROFILER
    ENABLE_VM_TRACING
    ENABLE_VM_OPCODE_PROFILER
    ENABLE_VM_STUB_LOGS
    ENABLE_SPATIAL_GRID_LOGS
    ENABLE_VM_EXCEPTIONS_LOGS
)
    if(${OPTION})
        target_compile_definitions(butterscotch PRIVATE ${OPTION})
    endif()
endforeach()

if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(butterscotch PROPERTIES
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"

        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_INFO_PLIST
            "${CMAKE_SOURCE_DIR}/src/ios/Info.plist.in"

        MACOSX_BUNDLE_GUI_IDENTIFIER
            "com.butterscotch.butterscotch"

        MACOSX_BUNDLE_BUNDLE_NAME "Butterscotch"
        MACOSX_BUNDLE_BUNDLE_VERSION "1"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0"

        XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME "AppIcon"
    )

    # iOS asset catalog
    set(IOS_ASSET_CATALOG "${CMAKE_BINARY_DIR}/ButterscotchAssets.xcassets")
    set(IOS_APP_ICON_SET "${IOS_ASSET_CATALOG}/AppIcon.appiconset")
    set(IOS_ICON_SET "${IOS_ASSET_CATALOG}/icon.imageset")

    file(MAKE_DIRECTORY
        "${IOS_APP_ICON_SET}"
        "${IOS_ICON_SET}"
    )

    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/res/logo.png")
        message(FATAL_ERROR "Missing iOS app icon: res/logo.png")
    endif()

    configure_file(
        "${CMAKE_SOURCE_DIR}/res/logo.png"
        "${IOS_APP_ICON_SET}/logo.png"
        COPYONLY
    )

    file(WRITE "${IOS_APP_ICON_SET}/Contents.json" [=[
    {
    "images": [
        {
        "filename": "logo.png",
        "idiom": "universal",
        "platform": "ios",
        "size": "1024x1024",
        "scale": "1x"
        }
    ],
    "info": {
        "author": "xcode",
        "version": 1
    }
    }
    ]=])

    if(EXISTS "${CMAKE_SOURCE_DIR}/res/icon.png")
        configure_file(
            "${CMAKE_SOURCE_DIR}/res/icon.png"

            COPYONLY
        )

        file(WRITE "${IOS_ICON_SET}/Contents.json" [=[
    {
    "images": [
        {
        "filename": "icon.png",
        "idiom": "universal",
        "scale": "1x"
        }
    ],
    "info": {
        "author": "xcode",
        "version": 1
    }
    }
    ]=])
    endif()

    target_sources(butterscotch PRIVATE "${IOS_ASSET_CATALOG}")

    set_source_files_properties(
        "${IOS_ASSET_CATALOG}"
        PROPERTIES MACOSX_PACKAGE_LOCATION "Resources"
    )
